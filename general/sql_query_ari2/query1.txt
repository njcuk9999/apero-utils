SELECT
    findex.KW_OBJNAME AS OBJNAME,
    astrom.ALIASES AS ALIASES,
    astrom.RA_DEG AS `RA [Deg]`,
    astrom.RA_SOURCE AS `RA source`,
    astrom.DEC_DEG AS `Dec [Deg]`,
    astrom.DEC_SOURCE AS `Dec source`,
    astrom.TEFF AS `Teff [K]`,
    astrom.TEFF_SOURCE AS `Teff source`,
    astrom.SP_TYPE AS SpT,
    astrom.SP_SOURCE AS `SpT source`,
    astrom.PMRA AS `PMRA [mas/yr]`,
    astrom.PMRA_SOURCE AS `PMRA source`,
    astrom.PMDE AS `PMDE [mas/yr]`,
    astrom.PMDE_SOURCE AS `PMDE source`,
    astrom.PLX AS `Plx [mas]`,
    astrom.PLX_SOURCE AS `Plx source`,
    astrom.RV AS `RV [km/s]`,
    astrom.RV_SOURCE AS `RV source`,
    findex.ALL_DPRTYPES AS DPRTYPE,
    findex.ALL_RUN_IDS AS RUN_ID,
    findex.ALL_PI_NAMES AS PI_NAMES,
    findex.raw_files AS `raw files`,
    findex.pp_files AS `pp files`,
    findex.pp_files_qcp AS `pp files [passed QC]`,
    findex.pp_files_qcf AS `pp files [failed QC]`,
    findex.pp_first_obs AS `pp files [first obs]`,
    findex.pp_latest_obs AS `pp files [latest obs]`,
    findex.pp_last_modified AS `pp files [last modified]`,
    findex.red_files AS `red files`,
    findex.red_files_qcp AS `red files [passed QC]`,
    findex.red_files_qcf AS `red files [failed QC]`,
    findex.red_first_obs AS `red files [first obs]`,
    findex.red_latest_obs AS `red files [latest obs]`,
    findex.red_last_modified AS `red files [last modified]`,
    findex.tcorr_files AS `tcorr files`,
    findex.tcorr_files_qcp AS `tcorr files [passed QC]`,
    findex.tcorr_files_qcf AS `tcorr files [failed QC]`,
    findex.tcorr_first_obs AS `tcorr files [first obs]`,
    findex.tcorr_latest_obs AS `tcorr files [latest obs]`,
    findex.tcorr_last_modified AS `tcorr files [last modified]`,
    findex.ccf_files AS `ccf files`,
    findex.ccf_files_qcp AS `ccf files [passed QC]`,
    findex.ccf_files_qcf AS `ccf files [failed QC]`,
    findex.ccf_first_obs AS `ccf files [first obs]`,
    findex.ccf_latest_obs AS `ccf files [latest obs]`,
    findex.ccf_last_modified AS `ccf files [last modified]`,
    findex.e_files_files AS `e.fits`,
    findex.e_files_files_qcp AS `e.fits [passed QC]`,
    findex.e_files_files_qcf AS `e.fits [failed QC]`,
    findex.e_files_first_obs AS `e.fits [first obs]`,
    findex.e_files_latest_obs AS `e.fits [latest obs]`,
    findex.e_files_last_modified AS `e.fits [last modified]`,
    findex.t_files_files AS `t.fits`,
    findex.t_files_files_qcp AS `t.fits [passed QC]`,
    findex.t_files_files_qcf AS `t.fits [failed QC]`,
    findex.t_files_first_obs AS `t.fits [first obs]`,
    findex.t_files_latest_obs AS `t.fits [latest obs]`,
    findex.t_files_last_modified AS `t.fits [last modified]`,
    findex.lbl_files AS `LBL`,
    findex.latest_obs AS `last obs`,
    findex.last_modified AS `last modified`
FROM (
    SELECT
        KW_OBJNAME,
        GROUP_CONCAT(DISTINCT fdb.KW_DPRTYPE SEPARATOR ", ") AS ALL_DPRTYPES,
        GROUP_CONCAT(DISTINCT fdb.KW_RUN_ID SEPARATOR ", ") AS ALL_RUN_IDS,
        GROUP_CONCAT(DISTINCT fdb.KW_PI_NAME SEPARATOR ", ") AS ALL_PI_NAMES,
        SUM(CASE WHEN fdb.BLOCK_KIND = "raw" THEN 1 ELSE 0 END) AS raw_files,
        SUM(CASE WHEN fdb.BLOCK_KIND = "tmp" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") THEN 1 ELSE 0 END) AS pp_files,
        SUM(CASE WHEN fdb.BLOCK_KIND = "tmp" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND ldb.PASSED_ALL_QC = 1 THEN 1 ELSE 0 END) AS pp_files_qcp,
        SUM(CASE WHEN fdb.BLOCK_KIND = "tmp" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND COALESCE(ldb.PASSED_ALL_QC, 0) = 0 THEN 1 ELSE 0 END) AS pp_files_qcf,
        FROM_UNIXTIME((MIN(CASE WHEN fdb.BLOCK_KIND = "tmp" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") THEN fdb.KW_MID_OBS_TIME END) - 40587) * 86400) AS pp_first_obs,
        FROM_UNIXTIME((MAX(CASE WHEN fdb.BLOCK_KIND = "tmp" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") THEN fdb.KW_MID_OBS_TIME END) - 40587) * 86400) AS pp_latest_obs,
        FROM_UNIXTIME(MAX(CASE WHEN fdb.BLOCK_KIND = "tmp" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") THEN fdb.LAST_MODIFIED END)) AS pp_last_modified,
        SUM(CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "EXT_E2DS_FF" AND fdb.KW_FIBER = "AB" THEN 1 ELSE 0 END) AS red_files,
        SUM(CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "EXT_E2DS_FF" AND fdb.KW_FIBER = "AB" AND ldb.PASSED_ALL_QC = 1 THEN 1 ELSE 0 END) AS red_files_qcp,
        SUM(CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "EXT_E2DS_FF" AND fdb.KW_FIBER = "AB" AND COALESCE(ldb.PASSED_ALL_QC, 0) = 0 THEN 1 ELSE 0 END) AS red_files_qcf,
        FROM_UNIXTIME((MIN(CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "EXT_E2DS_FF" AND fdb.KW_FIBER = "AB" THEN fdb.KW_MID_OBS_TIME END) - 40587) * 86400) AS red_first_obs,
        FROM_UNIXTIME((MAX(CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "EXT_E2DS_FF" AND fdb.KW_FIBER = "AB" THEN fdb.KW_MID_OBS_TIME END) - 40587) * 86400) AS red_latest_obs,
        FROM_UNIXTIME(MAX(CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "EXT_E2DS_FF" AND fdb.KW_FIBER = "AB" THEN fdb.LAST_MODIFIED END)) AS red_last_modified,
        SUM(CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "TELLU_OBJ" AND fdb.KW_FIBER = "AB" THEN 1 ELSE 0 END) AS tcorr_files,
        SUM(CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "TELLU_OBJ" AND fdb.KW_FIBER = "AB" AND ldb.PASSED_ALL_QC = 1 THEN 1 ELSE 0 END) AS tcorr_files_qcp,
        SUM(CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "TELLU_OBJ" AND fdb.KW_FIBER = "AB" AND COALESCE(ldb.PASSED_ALL_QC, 0) = 0 THEN 1 ELSE 0 END) AS tcorr_files_qcf,
        FROM_UNIXTIME((MIN(CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "TELLU_OBJ" AND fdb.KW_FIBER = "AB" THEN fdb.KW_MID_OBS_TIME END) - 40587) * 86400) AS tcorr_first_obs,
        FROM_UNIXTIME((MAX(CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "TELLU_OBJ" AND fdb.KW_FIBER = "AB" THEN fdb.KW_MID_OBS_TIME END) - 40587) * 86400) AS tcorr_latest_obs,
        FROM_UNIXTIME(MAX(CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "TELLU_OBJ" AND fdb.KW_FIBER = "AB" THEN fdb.LAST_MODIFIED END)) AS tcorr_last_modified,
        COUNT(DISTINCT CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "CCF_OBJ" AND fdb.KW_FIBER = "AB" THEN fdb.KW_PID END) AS ccf_files,
        COUNT(DISTINCT CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "CCF_OBJ" AND fdb.KW_FIBER = "AB" AND ldb.PASSED_ALL_QC = 1 THEN fdb.KW_PID END) AS ccf_files_qcp,
        COUNT(DISTINCT CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "CCF_OBJ" AND fdb.KW_FIBER = "AB" AND COALESCE(ldb.PASSED_ALL_QC, 0) = 0 THEN fdb.KW_PID END) AS ccf_files_qcf,
        FROM_UNIXTIME((MIN(CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "CCF_OBJ" AND fdb.KW_FIBER = "AB" THEN fdb.KW_MID_OBS_TIME END) - 40587) * 86400) AS ccf_first_obs,
        FROM_UNIXTIME((MAX(CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "CCF_OBJ" AND fdb.KW_FIBER = "AB" THEN fdb.KW_MID_OBS_TIME END) - 40587) * 86400) AS ccf_latest_obs,
        FROM_UNIXTIME(MAX(CASE WHEN fdb.BLOCK_KIND = "red" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "CCF_OBJ" AND fdb.KW_FIBER = "AB" THEN fdb.LAST_MODIFIED END)) AS ccf_last_modified,
        SUM(CASE WHEN fdb.BLOCK_KIND = "out" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "DRS_POST_E" AND fdb.KW_FIBER = "AB" THEN 1 ELSE 0 END) AS e_files_files,
        SUM(CASE WHEN fdb.BLOCK_KIND = "out" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "DRS_POST_E" AND fdb.KW_FIBER = "AB" AND ldb.PASSED_ALL_QC = 1 THEN 1 ELSE 0 END) AS e_files_files_qcp,
        SUM(CASE WHEN fdb.BLOCK_KIND = "out" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "DRS_POST_E" AND fdb.KW_FIBER = "AB" AND COALESCE(ldb.PASSED_ALL_QC, 0) = 0 THEN 1 ELSE 0 END) AS e_files_files_qcf,
        FROM_UNIXTIME((MIN(CASE WHEN fdb.BLOCK_KIND = "out" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "DRS_POST_E" AND fdb.KW_FIBER = "AB" THEN fdb.KW_MID_OBS_TIME END) - 40587) * 86400) AS e_files_first_obs,
        FROM_UNIXTIME((MAX(CASE WHEN fdb.BLOCK_KIND = "out" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "DRS_POST_E" AND fdb.KW_FIBER = "AB" THEN fdb.KW_MID_OBS_TIME END) - 40587) * 86400) AS e_files_latest_obs,
        FROM_UNIXTIME(MAX(CASE WHEN fdb.BLOCK_KIND = "out" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "DRS_POST_E" AND fdb.KW_FIBER = "AB" THEN fdb.LAST_MODIFIED END)) AS e_files_last_modified,
        SUM(CASE WHEN fdb.BLOCK_KIND = "out" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "DRS_POST_T" AND fdb.KW_FIBER = "AB" THEN 1 ELSE 0 END) AS t_files_files,
        SUM(CASE WHEN fdb.BLOCK_KIND = "out" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "DRS_POST_T" AND fdb.KW_FIBER = "AB" AND ldb.PASSED_ALL_QC = 1 THEN 1 ELSE 0 END) AS t_files_files_qcp,
        SUM(CASE WHEN fdb.BLOCK_KIND = "out" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "DRS_POST_T" AND fdb.KW_FIBER = "AB" AND COALESCE(ldb.PASSED_ALL_QC, 0) = 0 THEN 1 ELSE 0 END) AS t_files_files_qcf,
        FROM_UNIXTIME((MIN(CASE WHEN fdb.BLOCK_KIND = "out" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "DRS_POST_T" AND fdb.KW_FIBER = "AB" THEN fdb.KW_MID_OBS_TIME END) - 40587) * 86400) AS t_files_first_obs,
        FROM_UNIXTIME((MAX(CASE WHEN fdb.BLOCK_KIND = "out" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "DRS_POST_T" AND fdb.KW_FIBER = "AB" THEN fdb.KW_MID_OBS_TIME END) - 40587) * 86400) AS t_files_latest_obs,
        FROM_UNIXTIME(MAX(CASE WHEN fdb.BLOCK_KIND = "out" AND fdb.KW_DPRTYPE IN ("POLAR_FP","POLAR_DARK","OBJ_DARK","OBJ_FP") AND fdb.KW_OUTPUT = "DRS_POST_T" AND fdb.KW_FIBER = "AB" THEN fdb.LAST_MODIFIED END)) AS t_files_last_modified,
        SUM(CASE WHEN fdb.BLOCK_KIND = "lbl" AND fdb.KW_OUTPUT = "LBL_FITS" THEN 1 ELSE 0 END) AS lbl_files,
        FROM_UNIXTIME((MAX(fdb.KW_MID_OBS_TIME) - 40587) * 86400) AS latest_obs,
        FROM_UNIXTIME(MAX(fdb.LAST_MODIFIED)) AS last_modified
    FROM findex_spirou_offline_db fdb
    JOIN log_spirou_offline_db ldb ON fdb.KW_PID = ldb.PID
    GROUP BY KW_OBJNAME
) AS findex
JOIN astrom_spirou_offline_db astrom ON findex.KW_OBJNAME = astrom.OBJNAME;
